```typescript
export type ForceFieldKind = 'uniform' | 'gravity' | 'electric' | 'magnetic';
export interface BaseForceFieldComponent extends Component {
  type: 'forceField';
  kind: ForceFieldKind;
  position: [number, number, number];
  range: number;
}
export interface UniformFieldComponent extends BaseForceFieldComponent {
  kind: 'uniform';
  direction: [number, number, number];
  strength: number;
}
export interface GravityFieldComponent extends BaseForceFieldComponent {
  kind: 'gravity';
  strength: number;
  decay: boolean;
}
export interface ElectricFieldComponent extends BaseForceFieldComponent {
  kind: 'electric';
  charge: number;
  decay: boolean;
}
export interface MagneticFieldComponent extends BaseForceFieldComponent {
  kind: 'magnetic';
  direction: [number, number, number];
  strength: number;
}
export type ForceFieldComponent = UniformFieldComponent | GravityFieldComponent | ElectricFieldComponent | MagneticFieldComponent;
```

From RESEARCH.md §1:
- 均匀方向场: F = strength * direction
- 点引力源: F = -strength * r_vec / |r|^3 (decay=true) 或 F = -strength * r_hat (decay=false)
- 点电荷电场: E = k * Q * r_vec / |r|^3, F = q * E; k 缩放为 1 保证数值稳定
- 均匀磁场: F = q * (v x B), B = strength * direction_normalized
