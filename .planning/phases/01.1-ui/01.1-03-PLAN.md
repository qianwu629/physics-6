```tsx
import { type ReactNode } from 'react';

interface WorkspaceLayoutProps {
  menuBar: ReactNode;
  toolbox: ReactNode;
  canvas: ReactNode;
  inspector: ReactNode;
  toolbar: ReactNode;
  /** 浮动层 (ChartPanel, SceneBanner, SpringCreationBanner 等 z-overlay 元素) */
  overlays?: ReactNode;
}

export default function WorkspaceLayout({
  menuBar,
  toolbox,
  canvas,
  inspector,
  toolbar,
  overlays,
}: WorkspaceLayoutProps) {
  return (
    <div className="relative h-screen w-screen overflow-hidden">
      {/* === 主网格 === */}
      <div
        className="grid h-full w-full"
        style={{
          gridTemplateColumns: 'var(--sidebar-width) 1fr var(--inspector-width)',
          gridTemplateRows: 'var(--menubar-height) 1fr var(--toolbar-height)',
        }}
      >
        {/* MenuBar: 横跨全宽，第 1 行 */}
        <div className="col-span-3 row-start-1 z-[var(--z-panel)]">
          {menuBar}
        </div>

        {/* Toolbox: 左列，第 2 行 */}
        <div className="col-start-1 row-start-2 z-[var(--z-panel)] overflow-hidden">
          {toolbox}
        </div>

        {/* 3D Canvas: 中央，第 2 行 */}
        <div
          className="col-start-2 row-start-2 z-[var(--z-canvas)] relative"
          style={{ backgroundColor: 'var(--canvas-bg)' }}
        >
          {canvas}
        </div>

        {/* Inspector: 右列，第 2 行 */}
        <div className="col-start-3 row-start-2 z-[var(--z-panel)] overflow-hidden">
          {inspector}
        </div>

        {/* Toolbar: 横跨全宽，第 3 行 */}
        <div className="col-span-3 row-start-3 z-[var(--z-panel)]">
          {toolbar}
        </div>
      </div>

      {/* === 自由浮动层 (z-overlay: ChartPanel, Banners 等) === */}
      {overlays}
    </div>
  );
}
```

关键设计：
- 使用 `style={{ gridTemplateColumns }}` 而非 Tailwind 任意值类（回避 L-01: `var(--sidebar-width)` 被 Tailwind purge 风险）
- 每个网格区域 div 只负责定位和 z-index，不负责子组件的外观
- overlays slot 在主网格之外，z-overlay 层独立
