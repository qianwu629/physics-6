```tsx
import { Tabs, TabsList, TabsTrigger, TabsContent } from './ui/tabs';
import PropertyPanel from './PropertyPanel';
import EnvironmentPanel from './EnvironmentPanel';
import EntityList from './EntityList';
import ForceFieldPanelStub from './ForceFieldPanelStub';

export default function Inspector() {
  return (
    <div className="h-full flex flex-col overflow-hidden bg-[hsl(var(--card))] border-l border-[hsl(var(--border))]">
      <Tabs defaultValue="property" className="flex flex-col h-full">
        {/* Tab 栏 — 水平排列，固定在顶部 */}
        <TabsList className="shrink-0 w-full justify-start rounded-none border-b border-[hsl(var(--border))] bg-transparent h-10 px-2 gap-1">
          <TabsTrigger value="property" className="text-sm data-[state=active]:text-[hsl(var(--foreground))]">
            属性
          </TabsTrigger>
          <TabsTrigger value="environment" className="text-sm data-[state=active]:text-[hsl(var(--foreground))]">
            环境
          </TabsTrigger>
          <TabsTrigger value="entity" className="text-sm data-[state=active]:text-[hsl(var(--foreground))]">
            实体
          </TabsTrigger>
          <TabsTrigger value="forcefield" className="text-sm data-[state=active]:text-[hsl(var(--foreground))]" disabled>
            力场
          </TabsTrigger>
        </TabsList>

        {/* Tab 内容 — 占满剩余空间 */}
        <div className="flex-1 min-h-0 overflow-hidden">
          <TabsContent value="property" className="h-full m-0">
            <PropertyPanel />
          </TabsContent>
          <TabsContent value="environment" className="h-full m-0">
            <EnvironmentPanel />
          </TabsContent>
          <TabsContent value="entity" className="h-full m-0">
            <EntityList />
          </TabsContent>
          <TabsContent value="forcefield" className="h-full m-0">
            <ForceFieldPanelStub />
          </TabsContent>
        </div>
      </Tabs>
    </div>
  );
}
```

关键设计点：
- 背景使用 `bg-[hsl(var(--card))]` — 通过 CSS 变量主题切换时自动响应
- Tabs underline（active indicator）由 shadcn Tabs 的 `data-[state=active]` + border 样式控制 — 初始设置 border 在 ::after 伪元素，shadcn radix-nova preset 使用 `border-primary` 作为 active 指示色（符合 §4 Accent 白名单第 2 项）
- 「力场」Tab 使用 `disabled` 属性 — 真实 DOM 节点 + disabled 态（符合 §10 预留要求）
- `TabsContent` 使用 `h-full m-0` 消除默认 margin 且占满高度
- 不设置 `w-[320px]` — 宽度由父级 Inspector slot（WorkspaceLayout 的 col-start-3）通过 `var(--inspector-width)` 控制
