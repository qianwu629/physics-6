```typescript
/**
 * Chart FPS 基准测试
 *
 * 运行方式：在浏览器 dev console 中执行：
 *   import { runChartBenchmark } from './benchmark/chart-fps.ts';
 *   runChartBenchmark({ duration: 30000, entityCount: 4 });
 *
 * V-CHART-07: 16 条曲线 × 30 秒，median update cost < 3ms/frame
 */

interface BenchmarkOptions {
  duration?: number;      // 测试时长（毫秒），默认 30000
  entityCount?: number;   // 追踪实体数，默认 4
}

interface BenchmarkResult {
  medianUpdateMs: number;
  maxUpdateMs: number;
  minFps: number;
  avgFps: number;
  totalFrames: number;
  droppedFrames: number;
}

export function runChartBenchmark(options: BenchmarkOptions = {}): Promise<BenchmarkResult> {
  const { duration = 30000, entityCount = 4 } = options;

  return new Promise((resolve) => {
    const updateTimes: number[] = [];
    let frameCount = 0;
    let lastTime = performance.now();
    let droppedFrames = 0;
    const fpsSamples: number[] = [];

    // 模拟 chart.update() 调用（实际测试时替换为真实的 ChartCanvas refreshAll）
    const tick = () => {
      const now = performance.now();
      const elapsed = now - lastTime;
      frameCount++;

      // 检测掉帧（> 20ms 视为掉帧）
      if (elapsed > 20) {
        droppedFrames++;
      }

      // 每 500ms 采样一次 FPS
      if (frameCount % 30 === 0) {
        fpsSamples.push(Math.round(1000 / elapsed));
      }

      // 测量 update() 耗时
      const start = performance.now();
      // TODO: 调用 chartCanvasRef.current?.refreshAll()
      const end = performance.now();
      updateTimes.push(end - start);

      lastTime = now;

      if (now - start < duration) {
        requestAnimationFrame(tick);
      } else {
        const sorted = [...updateTimes].sort((a, b) => a - b);
        const median = sorted[Math.floor(sorted.length / 2)];
        const max = sorted[sorted.length - 1];
        const minFps = Math.min(...fpsSamples);
        const avgFps = Math.round(fpsSamples.reduce((a, b) => a + b, 0) / fpsSamples.length);

        resolve({
          medianUpdateMs: median,
          maxUpdateMs: max,
          minFps,
          avgFps,
          totalFrames: frameCount,
          droppedFrames,
        });
      }
    };

    requestAnimationFrame(tick);
  });
}
```

根据 checkpoint 反馈修复问题。常见问题及修复方案：

**问题 1: FPS 下降明显（V-CHART-07 失败）**
- 原因：ChartPanel 的 rAF 循环与 Scene3D 的 useFrame 竞争
- 修复：将 ChartPanel 的 refreshAll 调用从独立 rAF 改为与 Scene3D 共享的 useFrame（在 ChartSampler 中同时调用 refreshAll）
- 或者：降低图表刷新频率至 30Hz（每 2 帧更新一次）

**问题 2: 内存泄漏（10 分钟后 JS Heap 增长）**
- 原因：chartBuffers 未清理已删除实体的缓冲
- 修复：在 ChartSampler 的 useEffect([resetCounter]) 中调用 clearAllBuffers；在 entity 被删除时清理对应 buffer

**问题 3: 能量不守恒（V-CHART-01 失败）**
- 原因：computeEnergy 中 PE_gravity 符号错误或 PE_spring 计算遗漏
- 修复：检查公式符号，确认弹簧约束遍历完整

**问题 4: 加速度噪声过大（V-CHART-02 失败）**
- 原因：SMA 窗口太小或 dt 计算不准确
- 修复：增大 SMA 窗口至 7 或改用 EMA(α=0.3)

**问题 5: 时间窗口切换跳动（Pitfall #3）**
- 原因：setVisibleRange 和 update 混用导致
- 修复：确保时间窗口切换只调用 setVisibleRange，不修改数据

**问题 6: 拖拽/resize 卡顿**
- 原因：ResizeObserver 触发太频繁
- 修复：添加 throttle（16ms）或 debounce（100ms）

根据实际发现的问题选择性修复，不需要全部修复（可能没有问题）。
