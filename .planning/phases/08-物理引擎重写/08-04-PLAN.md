    ### 1. SpatialHash 类
    ```typescript
    export class SpatialHash {
      private cellSize: number;
      private cells: Map<string, Set<string>>;

      constructor(cellSize: number = 5) {
        this.cellSize = cellSize;
        this.cells = new Map();
      }

      private getCellKey(pos: { x: number; y: number; z: number }): string {
        const cx = Math.floor(pos.x / this.cellSize);
        const cy = Math.floor(pos.y / this.cellSize);
        const cz = Math.floor(pos.z / this.cellSize);
        return `${cx},${cy},${cz}`;
      }

      insert(entityId: string, pos: { x: number; y: number; z: number }): void {
        const key = this.getCellKey(pos);
        if (!this.cells.has(key)) {
          this.cells.set(key, new Set());
        }
        this.cells.get(key)!.add(entityId);
      }

      queryNeighbors(pos: { x: number; y: number; z: number }, range: number): string[] {
        const neighbors: string[] = [];
        const rangeInCells = Math.ceil(range / this.cellSize);
        const cx = Math.floor(pos.x / this.cellSize);
        const cy = Math.floor(pos.y / this.cellSize);
        const cz = Math.floor(pos.z / this.cellSize);

        for (let dx = -rangeInCells; dx <= rangeInCells; dx++) {
          for (let dy = -rangeInCells; dy <= rangeInCells; dy++) {
            for (let dz = -rangeInCells; dz <= rangeInCells; dz++) {
              const key = `${cx + dx},${cy + dy},${cz + dz}`;
              const cell = this.cells.get(key);
              if (cell) {
                neighbors.push(...cell);
              }
            }
          }
        }
        return neighbors;
      }

      clear(): void {
        this.cells.clear();
      }

      getCellCount(): number {
        return this.cells.size;
      }
    }
    ```

    ### 2. queryFieldSourceNeighbors 辅助函数
    ```typescript
    import type { Entity } from './types';

    export function queryFieldSourceNeighbors(
      spatialHash: SpatialHash,
      pos: { x: number; y: number; z: number },
      range: number,
      entities: Map<string, Entity>,
    ): string[] {
      const candidateIds = spatialHash.queryNeighbors(pos, range);
      // 过滤只保留有 fieldSource 组件的实体
      return candidateIds.filter(id => {
        const entity = entities.get(id);
        return entity?.components.has('fieldSource') ?? false;
      });
    }
    ```

    注意：
    - cellSize 默认 5m，与场景尺度匹配（实体通常分布在 10-50m 范围）
    - queryNeighbors 返回候选实体 ID 列表，调用方需进一步精确距离过滤
    - 空间哈希不替代距离计算，仅用于快速排除远距离实体
