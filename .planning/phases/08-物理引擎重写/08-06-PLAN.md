## 空间哈希集成点

可选优化：在 collectFieldSources 或 computeForceFromFieldSources 中使用空间哈希预过滤远距离场源。

由于 50 实体下 O(S×B) = 2500 次计算已足够快（<1ms），空间哈希作为可选优化，默认不启用。

集成方式：
```typescript
// 在 ForceFieldSystem 中可选使用
const spatialHash = useRef(new SpatialHash(5));

useBeforeStep(() => {
  // 每帧重建空间哈希
  spatialHash.current.clear();
  for (const [entityId, entity] of entities) {
    const src = entity.components.get('fieldSource');
    if (!src) continue;
    const ref = getRef(entityId)?.current;
    if (!ref) continue;
    const pos = ref.translation();
    spatialHash.current.insert(entityId, pos);
  }
  
  // 计算力时使用空间哈希过滤
  // ...
});
```

    ```typescript
      entities: Map<string, Entity>,
      getRef: (id: string) => RefObject<RigidBodyAPI | null> | undefined,
    ```

    ```typescript
      for (const [entityId, entity] of entities) {
        const src = entity.components.get('fieldSource') as FieldSourceComponent | undefined;
        if (!src) continue;
        const ref = getRef(entityId)?.current;
    ```

    ```typescript
    // frontend/src/ecs/__tests__/perf-fieldSource.bench.ts
    import { describe, bench } from 'vitest';
    import { collectFieldSources, computeForceFromFieldSources } from '../fieldSourceCalc';
    import type { Entity, FieldSourceComponent, RigidBodyComponent } from '../types';

    describe('fieldSource performance', () => {
      // 生成 N 个带 fieldSource 的实体
      function createMockEntities(count: number): Map<string, Entity> {
        const entities = new Map<string, Entity>();
        for (let i = 0; i < count; i++) {
          const entity: Entity = {
            id: `entity-${i}`,
            name: `Entity ${i}`,
            components: new Map([
              ['rigidBody', {
                type: 'rigidBody',
                kind: 'dynamic',
                mass: 1,
                restitution: 0.5,
                friction: 0.3,
                charge: (i % 3 === 0) ? 2 : 0, // 1/3 实体带电
              } as RigidBodyComponent],
              ...(i % 2 === 0 ? [['fieldSource', {
                type: 'fieldSource',
                kind: 'charge',
              } as FieldSourceComponent]] : []),
            ]),
          };
          entities.set(entity.id, entity);
        }
        return entities;
      }

      const mockGetRef = (id: string) => ({
        current: {
          translation: () => ({
            x: Math.random() * 50 - 25,
            y: Math.random() * 50 - 25,
            z: Math.random() * 50 - 25,
          }),
        },
      });

      bench('collectFieldSources — 10 entities', () => {
        const entities = createMockEntities(10);
        collectFieldSources(entities, mockGetRef, 0);
      });

      bench('collectFieldSources — 25 entities', () => {
        const entities = createMockEntities(25);
        collectFieldSources(entities, mockGetRef, 0);
      });

      bench('collectFieldSources — 50 entities', () => {
        const entities = createMockEntities(50);
        collectFieldSources(entities, mockGetRef, 0);
      });

      bench('computeForceFromFieldSources — 10 sources × 10 bodies', () => {
        const entities = createMockEntities(10);
        const sources = collectFieldSources(entities, mockGetRef, 0);
        for (let i = 0; i < 10; i++) {
          computeForceFromFieldSources(sources, { x: i, y: 0, z: 0 }, 1, `body-${i}`);
        }
      });

      bench('computeForceFromFieldSources — 25 sources × 25 bodies', () => {
        const entities = createMockEntities(25);
        const sources = collectFieldSources(entities, mockGetRef, 0);
        for (let i = 0; i < 25; i++) {
          computeForceFromFieldSources(sources, { x: i, y: 0, z: 0 }, 1, `body-${i}`);
        }
      });

      bench('computeForceFromFieldSources — 50 sources × 50 bodies', () => {
        const entities = createMockEntities(50);
        const sources = collectFieldSources(entities, mockGetRef, 0);
        for (let i = 0; i < 50; i++) {
          computeForceFromFieldSources(sources, { x: i, y: 0, z: 0 }, 1, `body-${i}`);
        }
      });
    });
    ```
