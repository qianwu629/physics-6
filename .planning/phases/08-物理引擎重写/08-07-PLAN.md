## 场源力线设计

### 电荷源电场线
- 正电荷：从电荷位置向外辐射（与现有 ElectricFieldComponent 正电荷逻辑一致）
- 负电荷：从边界向电荷位置汇聚（与现有 ElectricFieldComponent 负电荷逻辑一致）
- 线数量 ∝ |charge|（min 8, max 32）
- 颜色：正电荷红色 #ff4444，负电荷蓝色 #4444ff
- 使用 RigidBodyRefContext.getRef 获取实时位置（场源实体有 rigidBody）

### 电流源磁感线
- 无限长直导线简化：磁感线为以电流方向为轴的同心圆
- 在垂直于电流方向的平面上绘制同心圆
- 线数量 ∝ |current.magnitude|（min 8, max 32）
- 颜色：紫色 #a855f7（与磁场一致）
- 方向：右手定则（current_direction × r̂）

### 均匀磁场（现有）
- 已在 ForceFieldLines.tsx 中实现（magnetic kind 用平行线）
- 保持现有逻辑不变

    ### 3. 电荷源电场线生成函数
    ```typescript
    /**
     * 生成电荷源电场力线
     * 正电荷：从电荷位置向外辐射
     * 负电荷：从边界向电荷位置汇聚
     * 线数量 ∝ |charge|（min 8, max 32）
     */
    function generateChargeSourceLines(
      position: THREE.Vector3,
      charge: number,
      range: number = 20,  // 默认作用范围 20m
    ): { positions: Float32Array; colors: Float32Array } {
      const lineCount = Math.max(8, Math.min(MAX_LINES_ELECTRIC, Math.floor(Math.abs(charge) * 2)));
      const positions: number[] = [];
      const colors: number[] = [];

      const isPositive = charge > 0;
      const color = isPositive
        ? COLORS.electricPositive
        : charge < 0
          ? COLORS.electricNegative
          : COLORS.electricNeutral;

      const phi = Math.PI * (3 - Math.sqrt(5));

      for (let i = 0; i < lineCount; i++) {
        const y = 1 - (i / (lineCount - 1)) * 2;
        const radius = Math.sqrt(1 - y * y);
        const theta = phi * i;

        const dirX = Math.cos(theta) * radius;
        const dirY = y;
        const dirZ = Math.sin(theta) * radius;

        let start: THREE.Vector3;
        let end: THREE.Vector3;

        if (isPositive) {
          start = position.clone();
          end = new THREE.Vector3(
            position.x + dirX * range,
            position.y + dirY * range,
            position.z + dirZ * range,
          );
        } else {
          start = new THREE.Vector3(
            position.x + dirX * range,
            position.y + dirY * range,
            position.z + dirZ * range,
          );
          end = position.clone();
        }

        for (let s = 0; s <= SEGMENTS_PER_LINE; s++) {
          const t = s / SEGMENTS_PER_LINE;
          positions.push(
            start.x + (end.x - start.x) * t,
            start.y + (end.y - start.y) * t,
            start.z + (end.z - start.z) * t,
          );
          colors.push(color.r, color.g, color.b);
        }
      }

      return { positions: new Float32Array(positions), colors: new Float32Array(colors) };
    }
    ```

    ### 4. 电流源磁感线生成函数
    ```typescript
    /**
     * 生成电流源磁感线（无限长直导线简化）
     * 磁感线为以电流方向为轴的同心圆
     * 线数量 ∝ |current.magnitude|（min 8, max 32）
     */
    function generateCurrentSourceLines(
      position: THREE.Vector3,
      current: { magnitude: number; direction: [number, number, number] },
      range: number = 10,
    ): { positions: Float32Array; colors: Float32Array } {
      const lineCount = Math.max(8, Math.min(MAX_LINES_MAGNETIC, Math.floor(Math.abs(current.magnitude) * 2)));
      const positions: number[] = [];
      const colors: number[] = [];
      const color = COLORS.currentSource;

      const dir = new THREE.Vector3(...current.direction).normalize();
      const up = new THREE.Vector3(0, 1, 0);

      // 构建垂直于电流方向的局部坐标系
      let u = new THREE.Vector3().crossVectors(dir, up).normalize();
      if (u.lengthSq() < 0.001) {
        u = new THREE.Vector3(1, 0, 0);
      }
      const v = new THREE.Vector3().crossVectors(dir, u).normalize();

      // 绘制同心圆（在垂直于电流方向的平面上）
      for (let i = 0; i < lineCount; i++) {
        const radius = range * (0.2 + 0.8 * (i / (lineCount - 1)));  // 从 0.2*range 到 range
        const segments = SEGMENTS_PER_LINE * 2;  // 圆需要更多分段

        for (let s = 0; s <= segments; s++) {
          const angle = (s / segments) * Math.PI * 2;
          const localX = Math.cos(angle) * radius;
          const localY = Math.sin(angle) * radius;

          const point = position.clone()
            .add(u.clone().multiplyScalar(localX))
            .add(v.clone().multiplyScalar(localY));

          positions.push(point.x, point.y, point.z);
          colors.push(color.r, color.g, color.b);
        }
      }

      return { positions: new Float32Array(positions), colors: new Float32Array(colors) };
    }
    ```

    ### 5. 场源力线组件
    ```typescript
    function FieldSourceLineSegments({
      entityId,
      fieldSource,
      getRef,
    }: {
      entityId: string;
      fieldSource: FieldSourceComponent;
      getRef: (id: string) => { current?: { translation: () => { x: number; y: number; z: number } } | null } | undefined;
    }) {
      const geometry = useMemo(() => {
        const ref = getRef(entityId);
        const body = ref?.current;
        if (!body || typeof body.translation !== 'function') return null;

        const pos = body.translation();
        const position = new THREE.Vector3(pos.x, pos.y, pos.z);

        let data: { positions: Float32Array; colors: Float32Array } | null = null;

        if (fieldSource.kind === 'charge') {
          // 电荷源电场线 — 电荷值从 rigidBody 读取，此处仅传位置
          // 由于 FieldSourceComponent 不存储电荷值，力线密度用固定值
          // 实际电荷值在渲染时通过 entity 的 rigidBody 组件获取
          data = generateChargeSourceLines(position, 1, 20);
        } else if (fieldSource.kind === 'current' && fieldSource.current) {
          data = generateCurrentSourceLines(position, fieldSource.current, 10);
        }

        if (!data) return null;

        const geo = new THREE.BufferGeometry();
        geo.setAttribute('position', new THREE.BufferAttribute(data.positions, 3));
        geo.setAttribute('color', new THREE.BufferAttribute(data.colors, 3));
        return geo;
      }, [entityId, fieldSource, getRef]);

      useEffect(() => {
        return () => {
          geometry?.dispose();
        };
      }, [geometry]);

      if (!geometry) return null;

      return (
        <lineSegments geometry={geometry}>
          <lineBasicMaterial vertexColors transparent opacity={0.6} depthWrite={false} />
        </lineSegments>
      );
    }
    ```

    ### 6. 改造主组件 ForceFieldLines
    ```typescript
    export function ForceFieldLines() {
      const showForceLines = useVisualizationStore((s) => s.showForceLines);
      const showFieldSourceLines = useVisualizationStore((s) => s.showFieldSourceLines);  // 新增
      const entities = useSimulationStore((s) => s.entities);
      const { getRef } = useRigidBodyRefRegistry();  // 新增

      // 收集所有力场实体（现有逻辑）
      const forceFieldEntries: { id: string; field: ForceFieldComponent }[] = [];
      // 收集所有场源实体（新增）
      const fieldSourceEntries: { id: string; fieldSource: FieldSourceComponent }[] = [];

      for (const [id, entity] of entities) {
        const field = entity.components.get('forceField') as ForceFieldComponent | undefined;
        if (field) forceFieldEntries.push({ id, field });

        const fs = entity.components.get('fieldSource') as FieldSourceComponent | undefined;
        if (fs) fieldSourceEntries.push({ id, fieldSource: fs });
      }

      const hasForceFields = forceFieldEntries.length > 0;
      const hasFieldSources = fieldSourceEntries.length > 0;

      if ((!showForceLines || !hasForceFields) && (!showFieldSourceLines || !hasFieldSources)) {
        return null;
      }

      return (
        <group>
          {/* 预设力场力线（现有逻辑） */}
          {showForceLines && forceFieldEntries.map(({ id, field }) => (
            <FieldLineSegments key={`ff-${id}`} field={field} />
          ))}

          {/* 场源力线（新增） */}
          {showFieldSourceLines && fieldSourceEntries.map(({ id, fieldSource }) => (
            <FieldSourceLineSegments
              key={`fs-${id}`}
              entityId={id}
              fieldSource={fieldSource}
              getRef={getRef}
            />
          ))}
        </group>
      );
    }
    ```

    - 电流源磁感线密度：使用 current.magnitude 决定线数量
    - 场源力线与预设力场力线使用不同 key 前缀（fs- / ff-），避免 React key 冲突
    - 保持现有 ForceFieldLines 组件的导出签名不变
