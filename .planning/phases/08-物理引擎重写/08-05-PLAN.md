    ### 3. 事件处理函数
    在现有 handler 之后添加：
    ```typescript
    // 场源开关
    const handleToggleFieldSource = useCallback(
      (enabled: boolean) => {
        if (!selectedEntityId) return;
        if (enabled) {
          // 添加 fieldSource 组件（默认 charge 类型）
          updateComponent(selectedEntityId, 'fieldSource', {
            type: 'fieldSource',
            kind: 'charge',
            timeVarying: { enabled: false, amplitude: 0, frequency: 0, phase: 0 },
          });
        } else {
          // 删除 fieldSource 组件
          // 注意：updateComponent 不支持删除，需要通过 removeComponent 或设为空
          // 方案：使用 updateComponent 将 kind 设为 null 或添加 removeComponent 调用
          // 由于 entitySlice 可能没有 removeComponent，使用 updateComponent 设特殊标记
          // 或直接在 PropertyPanel 中调用 store 的 removeComponent 方法
        }
      },
      [selectedEntityId, updateComponent],
    );

    const handleFieldSourceKindChange = useCallback(
      (kind: FieldSourceKind) => {
        if (!selectedEntityId || !fieldSource) return;
        const update: Partial<FieldSourceComponent> = { kind };
        if (kind === 'current') {
          update.current = { magnitude: 1, direction: [0, 1, 0] };
        } else {
          update.current = undefined;
        }
        updateComponent(selectedEntityId, 'fieldSource', update);
      },
      [selectedEntityId, fieldSource, updateComponent],
    );

    const handleTimeVaryingToggle = useCallback(
      (enabled: boolean) => {
        if (!selectedEntityId || !fieldSource) return;
        updateComponent(selectedEntityId, 'fieldSource', {
          timeVarying: {
            enabled,
            amplitude: fieldSource.timeVarying?.amplitude ?? 1,
            frequency: fieldSource.timeVarying?.frequency ?? 1,
            phase: fieldSource.timeVarying?.phase ?? 0,
          },
        });
      },
      [selectedEntityId, fieldSource, updateComponent],
    );

    const handleTimeVaryingParamChange = useCallback(
      (key: 'amplitude' | 'frequency' | 'phase', val: number) => {
        if (!selectedEntityId || !fieldSource) return;
        updateComponent(selectedEntityId, 'fieldSource', {
          timeVarying: {
            enabled: fieldSource.timeVarying?.enabled ?? false,
            amplitude: fieldSource.timeVarying?.amplitude ?? 0,
            frequency: fieldSource.timeVarying?.frequency ?? 0,
            phase: fieldSource.timeVarying?.phase ?? 0,
            [key]: val,
          },
        });
      },
      [selectedEntityId, fieldSource, updateComponent],
    );

    const handleCurrentMagnitudeChange = useCallback(
      (val: number) => {
        if (!selectedEntityId || !fieldSource || fieldSource.kind !== 'current') return;
        updateComponent(selectedEntityId, 'fieldSource', {
          current: {
            magnitude: val,
            direction: fieldSource.current?.direction ?? [0, 1, 0],
          },
        });
      },
      [selectedEntityId, fieldSource, updateComponent],
    );

    const handleCurrentDirectionChange = useCallback(
      (index: number, val: number) => {
        if (!selectedEntityId || !fieldSource || fieldSource.kind !== 'current') return;
        const newDir: [number, number, number] = [
          ...(fieldSource.current?.direction ?? [0, 1, 0]),
        ];
        newDir[index] = val;
        updateComponent(selectedEntityId, 'fieldSource', {
          current: {
            magnitude: fieldSource.current?.magnitude ?? 1,
            direction: newDir,
          },
        });
      },
      [selectedEntityId, fieldSource, updateComponent],
    );
    ```

    在"物理参数"区域（电荷之后）添加场源折叠面板：

    ```tsx
    {/* 场源设置 — Phase 8 */}
    <div className="space-y-2">
      <div className="flex items-center justify-between">

          disabled={disabled}
        />
      </div>

      {fieldSource && (
        <div className="space-y-2 pl-2 border-l border-white/10">

            <span className="text-xs" style={{ color: '#888' }}>类型</span>
            <div className="flex gap-1">
              {(['charge', 'current'] as FieldSourceKind[]).map((k) => (
                <button
                  key={k}
                  type="button"
                  className={`px-2 py-0.5 text-xs rounded ${
                    fieldSource.kind === k
                      ? 'bg-blue-500/20 text-blue-400 border border-blue-500/30'

                  }`}
                  onClick={() => handleFieldSourceKindChange(k)}

                >

              ))}
            </div>
          </div>

          {/* 电荷源提示 */}

          {/* 电流源参数 */}

            <div className="space-y-2">

                min={0}
                max={100}

                disabled={disabled}
                onChange={handleCurrentMagnitudeChange}
              />
              <Vector3Field
                label="电流方向"
                value={fieldSource.current?.direction ?? [0, 1, 0]}
                min={-1}
                max={1}
                step={0.1}
                disabled={disabled}
                onChange={handleCurrentDirectionChange}
              />
            </div>
          )}

          {/* 时变参数 */}
          <div className="space-y-2 pt-1">

                onCheckedChange={handleTimeVaryingToggle}
                disabled={disabled}
              />
            </div>

              <div className="space-y-2">

                  value={fieldSource.timeVarying.amplitude}

                  min={0}
                  max={10}
                  step={0.1}
                  disabled={disabled}
                  onChange={(v) => handleTimeVaryingParamChange('amplitude', v)}
                />
                <PhysicsField
                  label="频率"
                  value={fieldSource.timeVarying.frequency}
                  unit="Hz"
                  min={0}
                  max={10}
                  step={0.1}
                  disabled={disabled}
                  onChange={(v) => handleTimeVaryingParamChange('frequency', v)}
                />
                <PhysicsField
                  label="初相位"
                  value={fieldSource.timeVarying.phase}
                  unit="rad"
                  min={0}
                  max={6.28}
                  step={0.1}
                  disabled={disabled}
                  onChange={(v) => handleTimeVaryingParamChange('phase', v)}
                />
              </div>
            )}
          </div>
        </div>
      )}
    </div>
    ```
