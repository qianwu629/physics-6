    export function getTimeVaryingValue(
      baseValue: number,
      timeVarying: FieldSourceComponent['timeVarying'],
      t: number,
    ): number {
      if (!timeVarying?.enabled) return baseValue;
      return baseValue + timeVarying.amplitude * Math.sin(2 * Math.PI * timeVarying.frequency * t + timeVarying.phase);
    }

    ### 5. computeElectricFieldFromSources
    计算所有电荷源在 targetPos 处产生的电场强度 E：
    ```typescript
    export function computeElectricFieldFromSources(
      sources: FieldSource[],
      targetPos: Vec3,
      excludeId?: string,
    ): Vec3
    ```
    算法：
    - 遍历 sources，跳过 excludeId（自相互作用）
    - 跳过 kind !== 'charge' 的源
    - 跳过 charge === 0 的源
    - 计算 r_vec = targetPos - source.position
    - r = |r_vec|，r > MAX_RANGE 跳过

    - Ex += Q × rx × inv_rSoft3（y, z 同理）
    - 返回 { x: COULOMB_K × Ex, y: COULOMB_K × Ey, z: COULOMB_K × Ez }

    ### 6. computeForceFromFieldSources

      bodyPos: Vec3,
      bodyCharge: number,
      bodyId: string,
    ): Vec3

    - bodyCharge === 0 → 返回 ZERO
    - E = computeElectricFieldFromSources(sources, bodyPos, bodyId)

    - 返回 F

    ```typescript
    /**
     * 合并电场和磁场对速度的影响。
     * 先用电场力计算速度增量，再对更新后的速度应用磁场旋转。
     * 避免 addForce 与 setLinvel 混用导致的积分器冲突。
     */
    export function updateVelocityWithEMField(
      body: {
        setLinvel(vel: Vec3, wakeUp: boolean): void;
        resetForces(wakeUp: boolean): void;
      },
      vel: Vec3,
      F_nonMag: Vec3,
      bodyB: Vec3,
      charge: number,
      mass: number,
      dt: number,
    ): Vec3 {
      if (charge === 0 || mass <= 0) return vel;

      // 1. 电场产生的速度增量（显式欧拉）
      const dv_electric = {
        x: (F_nonMag.x / mass) * dt,
        y: (F_nonMag.y / mass) * dt,
        z: (F_nonMag.z / mass) * dt,
      };

      // 2. 先应用电场速度变化
      const vel_after_electric = {
        x: vel.x + dv_electric.x,
        y: vel.y + dv_electric.y,
        z: vel.z + dv_electric.z,
      };

      // 3. 再应用磁场旋转（基于更新后的速度）
      const newVel = rotateVelocityByMagneticField(vel_after_electric, bodyB, charge, mass, dt);

      // 4. 清零旧力并设置新速度
      body.resetForces(true);
      body.setLinvel(newVel, true);

      return newVel;
    }
    ```
