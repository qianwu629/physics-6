```json
{
  "schemaVersion": "1.0",
  "savedAt": "2026-05-04T00:00:00Z",
  "simulation": {
    "environment": { "gravity": [0, -9.81, 0], "frictionScale": 1.0, "restitutionScale": 1.0, "drag": 0.1 },
    "entities": [
      {
        "id": "unique-id",
        "name": "显示名称",
        "components": {
          "transform": { "type": "transform", "position": [x, y, z], "rotation": [rx, ry, rz], "scale": [sx, sy, sz] },
          "rigidBody": { "type": "rigidBody", "kind": "dynamic|fixed", "mass": 1, "restitution": 0.5, "friction": 0.3 },
          "collider": { "type": "collider", "shape": "sphere|cuboid|cylinder", "params": { ... } },
          "material": { "type": "material", "color": "#3b82f6", "roughness": 0.6, "metalness": 0.1 },
          "velocity": { "type": "velocity", "linearVelocity": [vx, vy, vz], "angularVelocity": [0, 0, 0] }
        }
      }
    ],
    "constraints": [
      {
        "id": "spring-1",
        "name": "弹簧-1",
        "components": {
          "constraint": { "type": "constraint", "kind": "spring", "entityAId": "entity-id-1", "entityBId": "entity-id-2",
            "params": { "stiffness": 100, "restLength": 2.0, "damping": 0.1 } }
        }
      }
    ]
  }
}
```

注意：
- 预设 JSON 中不包含 `trail` 和 `vector` 组件（D-01-01），反序列化时 createEntity() 自动添加默认 trail(visible=true) + vector(showVelocity=true, showForces=true)
- constraints 数组独立于 entities，存储约束实体
- 约束实体的 entityAId/entityBId 必须引用 entities 数组中存在的实体 ID

```typescript
async function handlePresetClick(presetId: string) {
  // 1. 动态导入 JSON 文件
  const presetModule = await import(`../presets/${presetId}.json`);
  const presetData = presetModule.default || presetModule;

  // 2. 反序列化
  const result = deserializeScene(presetData);
  if (!result.success || !result.data) {
    // 错误理论上不应发生（预设 JSON 已验证），但保留处理
    alert(`预设加载失败: ${result.errors.join(', ')}`);
    return;
  }

  // 3. 关闭选择器
  onOpenChange(false);

  // 4. 通过确认+加载流程
  const loaded = await loadSceneWithConfirm(result.data);
  if (loaded) {
    // 加载成功——toast 由 SceneLoader 内部处理
    // warnings 由 SceneBanner 显示
  }
}
```
