      const v = { x, y, z } as any;
      v.clone = vi.fn(() => v);
      v.normalize = vi.fn(() => v);
      v.length = vi.fn(() => Math.sqrt(x*x + y*y + z*z));
      v.set = vi.fn((nx: number, ny: number, nz: number) => { v.x = nx; v.y = ny; v.z = nz; return v; });
      v.copy = vi.fn((src: any) => { v.x = src.x; v.y = src.y; v.z = src.z; return v; });
      v.add = vi.fn((other: any) => ({ x: v.x + other.x, y: v.y + other.y, z: v.z + other.z }));
      v.sub = vi.fn((other: any) => ({ x: v.x - other.x, y: v.y - other.y, z: v.z - other.z }));
      return v;
